The Automation Script

Create it with nano run_variant_calling.sh, paste the AI output, save with Ctrl+O → Enter → Ctrl+X.

bash
#!/bin/bash
# run_variant_calling.sh

module load bwa
module load samtools
module load bcftools

set -e

cd ~/dc_workshop

genome=~/dc_workshop/03_ref_genome/rel606.fasta
bwa index $genome

mkdir -p 04_alignments/sam 04_alignments/bam 04_alignments/bcf \
         05_genomic_variants/vcf 05_genomic_variants/filtered

for file in ~/dc_workshop/02_trimmed_reads/*.trim.fastq.gz; do

    echo "====================================="
    echo "Working with file: $file"
    base=$(basename ${file} .trim.fastq.gz)
    echo "Sample name: $base"

    sam=~/dc_workshop/04_alignments/sam/${base}.aligned.sam
    bam=~/dc_workshop/04_alignments/bam/${base}.aligned.bam
    sorted_bam=~/dc_workshop/04_alignments/bam/${base}.aligned.sorted.bam
    stats=~/dc_workshop/04_alignments/bam/${base}_flagstat.txt
    raw_bcf=~/dc_workshop/04_alignments/bcf/${base}_raw.bcf
    variants=~/dc_workshop/05_genomic_variants/vcf/${base}_variants.vcf
    final_variants=~/dc_workshop/05_genomic_variants/filtered/${base}_final_variants.vcf

    ## Align reads to reference genome
    bwa mem $genome $file > $sam

    ## Convert SAM to BAM
    samtools view -S -b $sam > $bam

    ## Sort BAM by genomic position
    samtools sort -o $sorted_bam $bam

    ## Index the sorted BAM
    samtools index $sorted_bam

    ## Save alignment statistics
    samtools flagstat $sorted_bam > $stats

    ## Calculate per-position read coverage
    bcftools mpileup -O b -o $raw_bcf -f $genome $sorted_bam

    ## Call SNP variants
    bcftools call --ploidy 1 -m -v -o $variants $raw_bcf

    ## Filter low-quality variants
    vcfutils.pl varFilter $variants > $final_variants

done

echo "Pipeline complete!"
Running the Script
bash
cd ~/dc_workshop

chmod u+x run_variant_calling.sh   # make executable

screen -S script                   # create a screen session
screen -ls                         # confirm you're attached

./run_variant_calling.sh           # run (~10 minutes for all 6 samples)

# Press Ctrl+A then D to detach while it runs
screen -r script                   # re-attach when done
