What the loop looks like

R
files <- list.files("data/vcfs")

combined_vcfs <- data.frame(matrix(ncol = 25))
names(combined_vcfs) <- names(vcf_data)

for (i in 1:length(files)) {
  file <- files[i]
  vcf <- read.vcfR(paste("data/vcfs/", file, sep = ""))
  vcf_clean <- vcfR2tidy(vcf)
  vcf_data <- vcf_clean$fix
  vcf_data$ID <- str_split(file, "_", simplify = T)[1]
  rm(vcf, vcf_clean, file)
  combined_vcfs <- rbind(combined_vcfs, vcf_data)
}

combined_vcfs <- combined_vcfs %>% filter(!is.na(CHROM))
table(combined_vcfs$ID)
